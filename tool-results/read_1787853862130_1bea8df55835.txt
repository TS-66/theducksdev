     1→/**

    51→const sleep = (ms: number, signal: AbortSignal) =>
    52→  new Promise<void>((resolve, reject) => {
    53→    const t = setTimeout(resolve, ms);
    54→    signal.addEventListener(
    55→      'abort',
    56→      () => {
    57→        clearTimeout(t);
    58→        reject(new DOMException('Aborted', 'AbortError'));
    59→      },
    60→      { once: true },
    61→    );
    62→  });

    64→/** Stream text with a lively cadence; aborts cleanly. */
    65→async function streamText(
    66→  text: string,
    67→  onEvent: (e: AgentEvent) => void,
    68→  signal: AbortSignal,
    69→  chunkSize = 6,
    70→  delayMs = 12,
    71→): Promise<void> {
    72→  for (let i = 0; i < text.length; i += chunkSize) {
    73→    if (signal.aborted) throw new DOMException('Aborted', 'AbortError');
    74→    onEvent({ type: 'text-delta', delta: text.slice(i, i + chunkSize) });
    75→    await sleep(delayMs, signal);
    76→  }
    77→}

    83→/**
    84→ * Recent web_search results per session, so a follow-up like
    85→ * “fetch the first result” can resolve against the REAL last search.

    89→
    90→const ORDINALS: Array<[RegExp, number]> = [
    91→  [/\b(?:first|top|1st|leading|best)\b/, 0],
    92→  [/\b(?:second|2nd)\b/, 1],
    93→  [/\b(?:third|3rd)\b/, 2],
    94→  [/\b(?:fourth|4th)\b/, 3],
    95→  [/\b(?:fifth|5th)\b/, 4],
    96→];
    97→
    98→function ordinalIndex(text: string): number {
    99→  for (const [re, idx] of ORDINALS) if (re.test(text)) return idx;
   100→  return -1;
   101→}
   102→
   103→function extractUrl(raw: string): string | null {
   104→  const m = raw.match(/https?:\/\/[^\s`'"<>）)]+/i);
   105→  if (!m) return null;
   106→  // trim trailing sentence punctuation that often glues onto URLs
   107→  return m[0].replace(/[.,;:!?…]+$/, '');
   108→}
   109→
   110→/**
   111→ * Pull the actual search topic out of a natural-language request:
   112→ * "search the web for ai models" → "ai models".
   113→ */
   114→function extractSearchQuery(raw: string): string {
   115→  let q = raw.trim();
   116→  q = q.replace(/^(please|can you|could you|ducky[,:]?|hey)\s+/i, '');
   117→  q = q.replace(/^(web\s*search|search\s+(the\s+web|online|the\s+internet))(\s+for|\s+about|\s+:)\s*/i, '');
   118→  q = q.replace(/^look\s*up(\s+on( the)? web)?(\s+for|\s+about)?\s*/i, '');
   119→  q = q.replace(/^search\s+for\s+/i, '');
   120→  q = q.replace(/^(for|about)\s+/i, '');
   121→  q = q.replace(/[?.!]+$/, '');
   122→  return q.length > 0 ? q : raw.trim();
   123→}

   472→        await streamText(
   473→          `Demo mode — but this one is **real**: \`web_search\` runs through the server-side plugin proxy and needs no model key. Searching for “${q}”…\n\n`,

   635→        let ok = true;
   636→        try {

   692→    }
   693→    emitSafe({ type: 'error', message: `Demo engine failure: ${(e as Error).message}` });
   694→    emitSafe({ type: 'done', aborted: false });
   695→  }
   696→}
