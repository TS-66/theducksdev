   434→/** Data tab — full backup export/import of every session as one JSON file. */
   435→function BackupSection() {
   436→  const sessions = useDshStore((s) => s.sessions);
   437→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   438→
   439→  const totalMessages = React.useMemo(
   440→    () => sessions.reduce((n, s) => n + s.messages.length, 0),
   441→    [sessions],
   442→  );
   443→  const totalFiles = React.useMemo(
   444→    () => sessions.reduce((n, s) => n + Object.keys(s.workspace).length, 0),
   445→    [sessions],
   446→  );
   447→
   448→  const exportBackup = () => {
   449→    if (sessions.length === 0) {
   450→      toast.info("Nothing to back up yet", { description: "Create a session first." });
   451→      return;
   452→    }
   453→    downloadSessionsBackup(sessions);
   454→    toast.success("Backup downloaded", {
   455→      description: `${sessions.length} session${sessions.length === 1 ? "" : "s"} · ${totalMessages} messages. API key never included.`,
   456→    });
   457→  };
   458→
   459→  const onImportFile = async (file: File) => {
   460→    try {
   461→      const res = parseBackupFile(await file.text());
   462→      const imported = useDshStore.getState().importSessions(res.sessions);
   463→      if (imported === 0) throw new Error("No valid sessions found in this backup.");
   464→      toast.success(`Imported ${imported} session${imported === 1 ? "" : "s"}`, {
   465→        description:
   466→          res.skipped > 0
   467→            ? `${res.skipped} malformed entr${res.skipped === 1 ? "y" : "ies"} skipped. Ids were regenerated — no collisions.`
   468→            : "Ids regenerated — safe to import repeatedly.",
   469→      });
   470→    } catch (e) {
   471→      toast.error("Import failed", {
   472→        description: e instanceof Error ? e.message : String(e),
   473→      });
   474→    }
   475→  };
   476→
   477→  return (
   478→    <section className="space-y-3 rounded-md border p-3">
   479→      <div className="flex items-center gap-2">
   480→        <span className="flex size-8 items-center justify-center rounded-md border border-[#4D6BFE]/40 bg-[#4D6BFE]/10">
   481→          <DatabaseBackup className="size-4 text-[#4D6BFE]" aria-hidden />
   482→        </span>
   483→        <div className="min-w-0">
   484→          <h3 className="text-xs font-semibold">Local backup</h3>
   485→          <p className="text-[11px] text-muted-foreground">
   486→            All sessions with messages, virtual workspaces and todos in one JSON file.
   487→          </p>
   488→        </div>
   489→      </div>
   490→
   491→      {/* stat strip */}
   492→      <div className="grid grid-cols-3 gap-2" aria-label="Backup scope">
   493→        {[
   494→          { label: "sessions", value: sessions.length },
   495→          { label: "messages", value: totalMessages },
   496→          { label: "v-files", value: totalFiles },
   497→        ].map((stat) => (
   498→          <div
   499→            key={stat.label}
   500→            className="rounded-md border border-border/60 bg-muted/30 px-2 py-1.5 text-center"
   501→          >
   502→            <p className="font-mono text-sm font-semibold">{stat.value}</p>
   503→            <p className="font-mono text-[9px] uppercase tracking-wide text-muted-foreground">
   504→              {stat.label}
   505→            </p>
   506→          </div>
   507→        ))}
   508→      </div>
   509→
   510→      <div className="grid grid-cols-2 gap-2">
   511→        <Button variant="outline" size="sm" onClick={exportBackup} disabled={sessions.length === 0}>
   512→          <Download className="mr-1.5 size-3.5" /> Export .json
   513→        </Button>
   514→        <Button
   515→          variant="outline"
   516→          size="sm"
   517→          onClick={() => fileInputRef.current?.click()}
   518→        >
   519→          <Upload className="mr-1.5 size-3.5" /> Import…
   520→        </Button>
   521→      </div>
   522→
   523→      <input
   524→        ref={fileInputRef}
   525→        type="file"
   526→        accept="application/json,.json"
   527→        className="hidden"
   528→        tabIndex={-1}
   529→        aria-hidden="true"
   530→        onChange={(e) => {
   531→          const f = e.target.files?.[0];
   532→          if (f) void onImportFile(f);
   533→          e.target.value = "";
   534→        }}
   535→      />
   536→
   537→      <ul className="space-y-1 border-t pt-2 text-[11px] leading-relaxed text-muted-foreground">
   538→        <li className="flex items-start gap-1.5">
   539→          <FileJson className="mt-0.5 size-3 shrink-0 text-muted-foreground/70" aria-hidden />
   540→          Imports are sanitized and ids regenerated, so re-importing never duplicates or collides.
   541→        </li>
   542→        <li className="flex items-start gap-1.5">
   543→          <TriangleAlert className="mt-0.5 size-3 shrink-0 text-amber-400" aria-hidden />
   544→          Your API key is deliberately never written to backups.
   545→        </li>
   546→      </ul>
   547→    </section>
   548→  );
   549→}

   553→const STORE_KEY = "dsh-web-store-v1";
   554→/** Chrome/Firefox grant ~5 MB per origin for localStorage — treat as the budget. */
   555→const LS_QUOTA = 5 * 1024 * 1024;
   556→
   557→function fmtBytes(n: number): string {
   558→  if (n >= 1024 * 1024) return `${(n / (1024 * 1024)).toFixed(2)} MB`;
   559→  if (n >= 1024) return `${(n / 1024).toFixed(1)} KB`;
   560→  return `${n} B`;
   561→}

   563→/**
   564→ * Live read of what dsh web actually occupies in localStorage, with a quota
   565→ * meter and the heaviest sessions called out — pasted images live here too.
   566→ */
   567→function StorageUsageSection() {
   568→  const sessions = useDshStore((s) => s.sessions);
   569→
   570→  const usage = React.useMemo(() => {
   571→    try {
   572→      const raw = localStorage.getItem(STORE_KEY) ?? "";
   573→      const total = new Blob([raw]).size;
   574→      let persisted: Array<Record<string, unknown>> = [];
   575→      try {
   576→        persisted =
   577→          (JSON.parse(raw) as { state?: { sessions?: Array<Record<string, unknown>> } })
   578→            .state?.sessions ?? [];
   579→      } catch {
   580→        /* degraded measurements only */
   581→      }
   582→      const byId = new Map(
   583→        persisted.map((p) => [String(p.id ?? ""), p] as const),
   584→      );
   585→      const rows = sessions
   586→        .map((s) => ({
   587→          id: s.id,
   588→          title: s.title,
   589→          bytes: (() => {
   590→            const p = byId.get(s.id);
   591→            return p ? new Blob([JSON.stringify(p)]).size : 0;
   592→          })(),
   593→          files: Object.keys(s.workspace).length,
   594→          images: Object.values(s.workspace).filter((c) => typeof c === "string" && isDataUrlImage(c)).length,
   595→        }))
   596→        .sort((a, b) => b.bytes - a.bytes);
   597→      return { total, rows };
   598→    } catch {
   599→      return { total: 0, rows: [] };
   600→    }
   601→  }, [sessions]);
   602→
   603→  const pct = Math.min(100, (usage.total / LS_QUOTA) * 100);
   604→  const tone = pct >= 85 ? "bg-red-500" : pct >= 60 ? "bg-amber-500" : "bg-emerald-500";
   605→  const topRows = usage.rows.filter((r) => r.bytes > 0).slice(0, 4);
   606→
   607→  return (
   608→    <section className="space-y-3 rounded-md border p-3">
   609→      <div className="flex items-center gap-2">
   610→        <span className="flex size-8 items-center justify-center rounded-md border border-violet-500/40 bg-violet-500/10">
   611→          <HardDrive className="size-4 text-violet-400" aria-hidden />
   612→        </span>
   613→        <div className="min-w-0">
   614→          <h3 className="text-xs font-semibold">Storage usage</h3>
   615→          <p className="text-[11px] text-muted-foreground">
   616→            Everything lives in this browser&apos;s localStorage — messages, workspaces and pasted
   617→            images.
   618→          </p>
   619→        </div>
   620→      </div>
   621→
   622→      {/* quota meter */}
   623→      <div>
   624→        <div className="mb-1 flex items-baseline justify-between font-mono text-[10px] text-muted-foreground">
   625→          <span>
   626→            {fmtBytes(usage.total)} used · {fmtBytes(Math.max(0, LS_QUOTA - usage.total))} free
   627→          </span>
   628→          <span>{pct.toFixed(pct < 10 ? 1 : 0)}%</span>
   629→        </div>
   630→        <div
   631→          role="progressbar"
   632→          aria-valuenow={Math.round(pct)}
   633→          aria-valuemin={0}
   634→          aria-valuemax={100}
   635→          aria-label="localStorage quota used"
   636→          className="h-1.5 overflow-hidden rounded-full bg-muted"
   637→        >
   638→          <div
   639→            className={cn("h-full rounded-full transition-all duration-500", tone)}
   640→            style={{ width: `${Math.max(pct, usage.total > 0 ? 2 : 0)}%` }}
   641→          />
   642→        </div>
   643→      </div>
   644→
   645→      {topRows.length > 0 ? (
   646→        <ul className="space-y-1 border-t pt-2">
   647→          {topRows.map((r) => (
   648→            <li key={r.id} className="flex items-center gap-2 font-mono text-[10px]">
   649→              <span className="min-w-0 flex-1 truncate text-muted-foreground">{r.title}</span>
   650→              {r.images > 0 && (
   651→                <span
   652→                  className="flex shrink-0 items-center gap-0.5 rounded bg-pink-500/10 px-1 py-px text-pink-300"
   653→                  title={`${r.images} image${r.images === 1 ? "" : "s"} in workspace`}
   654→                >
   655→                  <ImageIcon className="size-2.5" aria-hidden /> {r.images}
   656→                </span>
   657→              )}
   658→              <span className="w-16 shrink-0 text-right text-foreground/80">{fmtBytes(r.bytes)}</span>
   659→            </li>
   660→          ))}
   661→        </ul>
   662→      ) : (
   663→        <p className="border-t pt-2 font-mono text-[10px] text-muted-foreground">
   664→          no measurable sessions yet.
   665→        </p>
   666→      )}
   667→
   668→      {pct >= 60 && (
   669→        <p className="flex items-start gap-1.5 text-[11px] leading-relaxed text-amber-400">
   670→          <TriangleAlert className="mt-0.5 size-3 shrink-0" aria-hidden />
   671→          Storage is filling up — export a backup, then delete heavy sessions (large pasted images
   672→          are usually the culprit).
   673→        </p>
   674→      )}
   675→    </section>
   676→  );
   677→}

   680→  return /^data:image\//i.test(content.slice(0, 40));
