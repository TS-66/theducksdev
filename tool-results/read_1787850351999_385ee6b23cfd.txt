   437→/** Data tab — full backup export/import of every session as one JSON file. */
   438→function BackupSection() {
   439→  const sessions = useDuckyStore((s) => s.sessions);
   440→  const fileInputRef = React.useRef<HTMLInputElement>(null);
   441→
   442→  const totalMessages = React.useMemo(
   443→    () => sessions.reduce((n, s) => n + s.messages.length, 0),
   444→    [sessions],
   445→  );
   446→  const totalFiles = React.useMemo(
   447→    () => sessions.reduce((n, s) => n + Object.keys(s.workspace).length, 0),
   448→    [sessions],
   449→  );
   450→
   451→  const exportBackup = () => {
   452→    if (sessions.length === 0) {
   453→      toast.info("Nothing to back up yet", { description: "Create a session first." });
   454→      return;
   455→    }
   456→    downloadSessionsBackup(sessions);
   457→    toast.success("Backup downloaded", {
   458→      description: `${sessions.length} session${sessions.length === 1 ? "" : "s"} · ${totalMessages} messages. API key never included.`,
   459→    });
   460→  };
   461→
   462→  const onImportFile = async (file: File) => {
   463→    try {
   464→      const res = parseBackupFile(await file.text());
   465→      const imported = useDuckyStore.getState().importSessions(res.sessions);
   466→      if (imported === 0) throw new Error("No valid sessions found in this backup.");
   467→      toast.success(`Imported ${imported} session${imported === 1 ? "" : "s"}`, {
   468→        description:
   469→          res.skipped > 0
   470→            ? `${res.skipped} malformed entr${res.skipped === 1 ? "y" : "ies"} skipped. Ids were regenerated — no collisions.`
   471→            : "Ids regenerated — safe to import repeatedly.",
   472→      });
   473→    } catch (e) {
   474→      toast.error("Import failed", {
   475→        description: e instanceof Error ? e.message : String(e),
   476→      });
   477→    }
   478→  };
   479→
   480→  return (
   481→    <section className="space-y-3 rounded-md border p-3">
   482→      <div className="flex items-center gap-2">
   483→        <span className="flex size-8 items-center justify-center rounded-md border border-[#FDC00A]/40 bg-[#FDC00A]/10">
   484→          <DatabaseBackup className="size-4 text-[#FDC00A]" aria-hidden />
   485→        </span>
   486→        <div className="min-w-0">
   487→          <h3 className="text-xs font-semibold">Local backup</h3>
   488→          <p className="text-[11px] text-muted-foreground">
   489→            All sessions with messages, virtual workspaces and todos in one JSON file.
   490→          </p>
   491→        </div>
   492→      </div>
   493→
   494→      {/* stat strip */}
   495→      <div className="grid grid-cols-3 gap-2" aria-label="Backup scope">
   496→        {[
   497→          { label: "sessions", value: sessions.length },
   498→          { label: "messages", value: totalMessages },
   499→          { label: "v-files", value: totalFiles },
   500→        ].map((stat) => (
   501→          <div
   502→            key={stat.label}
   503→            className="rounded-md border border-border/60 bg-muted/30 px-2 py-1.5 text-center"
   504→          >
   505→            <p className="font-mono text-sm font-semibold">{stat.value}</p>
   506→            <p className="font-mono text-[9px] uppercase tracking-wide text-muted-foreground">
   507→              {stat.label}
   508→            </p>
   509→          </div>
   510→        ))}
   511→      </div>
   512→
   513→      <div className="grid grid-cols-2 gap-2">
   514→        <Button variant="outline" size="sm" onClick={exportBackup} disabled={sessions.length === 0}>
   515→          <Download className="mr-1.5 size-3.5" /> Export .json
   516→        </Button>
   517→        <Button
   518→          variant="outline"
   519→          size="sm"
   520→          onClick={() => fileInputRef.current?.click()}
   521→        >
   522→          <Upload className="mr-1.5 size-3.5" /> Import…
   523→        </Button>
   524→      </div>
   525→
   526→      <input
   527→        ref={fileInputRef}
   528→        type="file"
   529→        accept="application/json,.json"
   530→        className="hidden"
   531→        tabIndex={-1}
   532→        aria-hidden="true"
   533→        onChange={(e) => {
   534→          const f = e.target.files?.[0];
   535→          if (f) void onImportFile(f);
   536→          e.target.value = "";
   537→        }}
   538→      />
   539→
   540→      <ul className="space-y-1 border-t pt-2 text-[11px] leading-relaxed text-muted-foreground">
   541→        <li className="flex items-start gap-1.5">
   542→          <FileJson className="mt-0.5 size-3 shrink-0 text-muted-foreground/70" aria-hidden />
   543→          Imports are sanitized and ids regenerated, so re-importing never duplicates or collides.
   544→        </li>
   545→        <li className="flex items-start gap-1.5">
   546→          <TriangleAlert className="mt-0.5 size-3 shrink-0 text-amber-400" aria-hidden />
   547→          Your API key is deliberately never written to backups.
   548→        </li>
   549→      </ul>
   550→    </section>
   551→  );

   554→/* ───────────────────────── storage usage (Data tab) ─────────────────────── */
   555→
   556→const STORE_KEY = "ducky-coder-store-v1";
   557→/** Chrome/Firefox grant ~5 MB per origin for localStorage — treat as the budget. */
   558→const LS_QUOTA = 5 * 1024 * 1024;
   559→
   560→function fmtBytes(n: number): string {
   561→  if (n >= 1024 * 1024) return `${(n / (1024 * 1024)).toFixed(2)} MB`;
   562→  if (n >= 1024) return `${(n / 1024).toFixed(1)} KB`;
   563→  return `${n} B`;
   564→}
   565→
   566→/**
   567→ * Live read of what Ducky Coder actually occupies in localStorage, with a quota
   568→ * meter and the heaviest sessions called out — pasted images live here too.
   569→ */
   570→function StorageUsageSection() {
   571→  const sessions = useDuckyStore((s) => s.sessions);
   572→
   573→  const usage = React.useMemo(() => {
   574→    try {
   575→      const raw = localStorage.getItem(STORE_KEY) ?? "";
   576→      const total = new Blob([raw]).size;
   577→      let persisted: Array<Record<string, unknown>> = [];
   578→      try {
   579→        persisted =
   580→          (JSON.parse(raw) as { state?: { sessions?: Array<Record<string, unknown>> } })
   581→            .state?.sessions ?? [];
   582→      } catch {
   583→        /* degraded measurements only */
   584→      }
   585→      const byId = new Map(
   586→        persisted.map((p) => [String(p.id ?? ""), p] as const),
   587→      );
   588→      const rows = sessions
   589→        .map((s) => ({
   590→          id: s.id,
   591→          title: s.title,
   592→          bytes: (() => {
   593→            const p = byId.get(s.id);
   594→            return p ? new Blob([JSON.stringify(p)]).size : 0;
   595→          })(),
   596→          files: Object.keys(s.workspace).length,
   597→          images: Object.values(s.workspace).filter((c) => typeof c === "string" && isDataUrlImage(c)).length,
   598→        }))
   599→        .sort((a, b) => b.bytes - a.bytes);
   600→      return { total, rows };
   601→    } catch {
   602→      return { total: 0, rows: [] };
   603→    }
   604→  }, [sessions]);
   605→
   606→  const pct = Math.min(100, (usage.total / LS_QUOTA) * 100);
   607→  const tone = pct >= 85 ? "bg-red-500" : pct >= 60 ? "bg-amber-500" : "bg-emerald-500";
   608→  const topRows = usage.rows.filter((r) => r.bytes > 0).slice(0, 4);
   609→
   610→  return (
   611→    <section className="space-y-3 rounded-md border p-3">
   612→      <div className="flex items-center gap-2">
   613→        <span className="flex size-8 items-center justify-center rounded-md border border-violet-500/40 bg-violet-500/10">
   614→          <HardDrive className="size-4 text-violet-400" aria-hidden />
   615→        </span>
   616→        <div className="min-w-0">
   617→          <h3 className="text-xs font-semibold">Storage usage</h3>
   618→          <p className="text-[11px] text-muted-foreground">
   619→            Everything lives in this browser&apos;s localStorage — messages, workspaces and pasted
   620→            images.
   621→          </p>
   622→        </div>
   623→      </div>
   624→
   625→      {/* quota meter */}
   626→      <div>
   627→        <div className="mb-1 flex items-baseline justify-between font-mono text-[10px] text-muted-foreground">
   628→          <span>
   629→            {fmtBytes(usage.total)} used · {fmtBytes(Math.max(0, LS_QUOTA - usage.total))} free
   630→          </span>
   631→          <span>{pct.toFixed(pct < 10 ? 1 : 0)}%</span>
   632→        </div>
   633→        <div
   634→          role="progressbar"
   635→          aria-valuenow={Math.round(pct)}
   636→          aria-valuemin={0}
   637→          aria-valuemax={100}
   638→          aria-label="localStorage quota used"
   639→          className="h-1.5 overflow-hidden rounded-full bg-muted"
   640→        >
   641→          <div
   642→            className={cn("h-full rounded-full transition-all duration-500", tone)}
   643→            style={{ width: `${Math.max(pct, usage.total > 0 ? 2 : 0)}%` }}
   644→          />
   645→        </div>
   646→      </div>
   647→
   648→      {topRows.length > 0 ? (
   649→        <ul className="space-y-1 border-t pt-2">
   650→          {topRows.map((r) => (
   651→            <li key={r.id} className="flex items-center gap-2 font-mono text-[10px]">
   652→              <span className="min-w-0 flex-1 truncate text-muted-foreground">{r.title}</span>
   653→              {r.images > 0 && (
   654→                <span
   655→                  className="flex shrink-0 items-center gap-0.5 rounded bg-pink-500/10 px-1 py-px text-pink-300"
   656→                  title={`${r.images} image${r.images === 1 ? "" : "s"} in workspace`}
   657→                >
   658→                  <ImageIcon className="size-2.5" aria-hidden /> {r.images}
   659→                </span>
   660→              )}
   661→              <span className="w-16 shrink-0 text-right text-foreground/80">{fmtBytes(r.bytes)}</span>
   662→            </li>
   663→          ))}
   664→        </ul>
   665→      ) : (
   666→        <p className="border-t pt-2 font-mono text-[10px] text-muted-foreground">
   667→          no measurable sessions yet.
   668→        </p>
   669→      )}
   670→
   671→      {pct >= 60 && (
   672→        <p className="flex items-start gap-1.5 text-[11px] leading-relaxed text-amber-400">
   673→          <TriangleAlert className="mt-0.5 size-3 shrink-0" aria-hidden />
   674→          Storage is filling up — export a backup, then delete heavy sessions (large pasted images
   675→          are usually the culprit).
   676→        </p>
   677→      )}
   678→    </section>
   679→  );
   680→}
