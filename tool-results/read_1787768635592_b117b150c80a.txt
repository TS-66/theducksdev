   149→function fmtDur(ms?: number): string | null {
   150→  if (typeof ms !== "number") return null;
   151→  return ms >= 1000 ? `${(ms / 1000).toFixed(1)}s` : `${Math.round(ms)}ms`;

   154→type LedgerFilter = "all" | "prompts" | "replies" | "tools" | "files";
   155→
   156→const LEDGER_FILTERS: Array<{ id: LedgerFilter; label: string }> = [
   157→  { id: "all", label: "all" },
   158→  { id: "prompts", label: "prompts" },
   159→  { id: "replies", label: "replies" },
   160→  { id: "tools", label: "tools" },
   161→  { id: "files", label: "files" },

   163→
   164→function entryMatchesFilter(e: TimelineEntry, f: LedgerFilter): boolean {
   165→  switch (f) {
   166→    case "all":
   167→      return true;
   168→    case "prompts":
   169→      return e.kind === "user";
   170→    case "replies":
   171→      return e.kind === "assistant";
   172→    case "tools":
   173→      return e.kind === "tool" || e.kind === "plan" || e.kind === "todo";
   174→    case "files":
   175→      return (e.files?.length ?? 0) > 0;
   176→  }
   177→}

   183→function LedgerBrowser({
   184→  session,
   185→  onPreviewFile,
   186→  running,
   187→}: {
   188→  session: Session;
   189→  onPreviewFile: (path: string) => void;
   190→  running: boolean;
   191→}) {
   192→  const entries = React.useMemo(() => buildTimeline(session), [session]);
   193→  const [filter, setFilter] = React.useState<LedgerFilter>("all");
   194→  const [query, setQuery] = React.useState("");
   195→
   196→  const counts = React.useMemo(() => {
   197→    const c: Record<LedgerFilter, number> = {
   198→      all: entries.length,
   199→      prompts: 0,
   200→      replies: 0,
   201→      tools: 0,
   202→      files: 0,
   203→    };
   204→    for (const e of entries) {
   205→      if (e.kind === "user") c.prompts++;
   206→      else if (e.kind === "assistant") c.replies++;
   207→      else if (e.kind === "tool" || e.kind === "plan" || e.kind === "todo") c.tools++;
   208→      if ((e.files?.length ?? 0) > 0) c.files++;
   209→    }
   210→    return c;
   211→  }, [entries]);
   212→
   213→  const filtered = React.useMemo(() => {
   214→    const q = query.trim().toLowerCase();
   215→    return entries.filter((e) => {
   216→      if (!entryMatchesFilter(e, filter)) return false;
   217→      if (!q) return true;
   218→      return (
   219→        e.title.toLowerCase().includes(q) ||
   220→        (e.detail ?? "").toLowerCase().includes(q) ||
   221→        (e.toolName ?? "").toLowerCase().includes(q) ||
   222→        (e.files ?? []).some((f) => f.toLowerCase().includes(q))
   223→      );
   224→    });
   225→  }, [entries, filter, query]);
   226→
   227→  return (
   228→    <>
   229→      <LedgerSummaryStrip session={session} entries={entries} />
   230→      <div
   231→        className="mb-2 flex flex-wrap items-center gap-1"
   232→        role="group"
   233→        aria-label="Filter ledger entries"
   234→      >
   235→        {LEDGER_FILTERS.map((f) => {
   236→          const selected = filter === f.id;
   237→          return (
   238→            <button
   239→              key={f.id}
   240→              type="button"
   241→              aria-pressed={selected}
   242→              onClick={() => setFilter(f.id)}
   243→              className={cn(
   244→                "rounded-full border px-2 py-0.5 font-mono text-[10px] transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring",
   245→                selected
   246→                  ? "border-[#4D6BFE]/60 bg-[#4D6BFE]/15 text-[#9dabff]"
   247→                  : "border-border/60 bg-muted/20 text-muted-foreground hover:border-border hover:bg-muted/50 hover:text-foreground",
   248→              )}
   249→            >
   250→              {f.label}
   251→              <span className={cn("ml-1", selected ? "text-[#4D6BFE]" : "text-muted-foreground/60")}>
   252→                {counts[f.id]}
   253→              </span>
   254→            </button>
   255→          );
   256→        })}
   257→      </div>
   258→      <div className="relative mb-3">
   259→        <Search
   260→          className="pointer-events-none absolute left-2 top-1/2 size-3 -translate-y-1/2 text-muted-foreground/60"
   261→          aria-hidden
   262→        />
   263→        <Input
   264→          value={query}
   265→          onChange={(e) => setQuery(e.target.value)}
   266→          placeholder="grep the ledger… (title, tool, path)"
   267→          aria-label="Search ledger entries"
   268→          className="h-7 border-border/60 bg-muted/20 pl-7 pr-7 font-mono text-[11px] placeholder:text-muted-foreground/50 focus-visible:ring-1"
   269→        />
   270→        {query && (
   271→          <button
   272→            type="button"
   273→            aria-label="Clear ledger search"
   274→            onClick={() => setQuery("")}
   275→            className="absolute right-1.5 top-1/2 -translate-y-1/2 rounded-sm p-0.5 text-muted-foreground/70 transition-colors hover:bg-muted hover:text-foreground focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
   276→          >
   277→            <X className="size-3" aria-hidden />
   278→          </button>
   279→        )}
   280→      </div>
   281→      {filtered.length === 0 ? (
   282→        <p className="py-10 text-center font-mono text-xs text-muted-foreground">
   283→          {entries.length === 0
   284→            ? "no entries yet."
   285→            : query
   286→              ? `no matches for “${query}” under “${filter}”.`
   287→              : `no entries match “${filter}”.`}
   288→        </p>
   289→      ) : (
   290→        <LedgerView
   291→          entries={filtered}
   292→          session={session}
   293→          onPreviewFile={onPreviewFile}
   294→          running={running}
   295→        />
   296→      )}
   297→    </>
   298→  );
   299→}

   301→/** git-log diffstat vibes: one quiet strip of what this session's ledger holds. */
   302→function LedgerSummaryStrip({
   303→  session,
   304→  entries,
   305→}: {
   306→  session: Session;
   307→  entries: TimelineEntry[];
   308→}) {
   309→  const toolCalls = entries.filter((e) => e.kind === "tool").length;
   310→  const files = new Set<string>();
   311→  for (const e of entries) for (const f of e.files ?? []) files.add(f);
   312→  const tokens = (session.stats?.promptTokens ?? 0) + (session.stats?.completionTokens ?? 0);
   313→
   314→  const stats = [
   315→    { label: "entries", value: String(entries.length), glyph: "≡" },
   316→    { label: "tool calls", value: String(toolCalls), glyph: "⚙" },
   317→    { label: "files", value: String(files.size), glyph: "◇" },
   318→    { label: "tokens", value: tokens > 0 ? fmtK(tokens) : "—", glyph: "⚡" },
   319→  ];
   320→
   321→  return (
   322→    <div
   323→      className="mb-3 grid grid-cols-4 gap-1.5"
   324→      aria-label="Ledger summary"
   325→    >
   326→      {stats.map((s) => (
   327→        <div
   328→          key={s.label}
   329→          className="rounded-md border border-border/60 bg-muted/20 px-2 py-1.5 text-center transition-colors hover:border-border hover:bg-muted/40"
   330→        >
   331→          <p className="font-mono text-[13px] font-semibold leading-tight">
   332→            <span className="mr-1 text-[10px] text-muted-foreground/70" aria-hidden>
   333→              {s.glyph}
   334→            </span>
   335→            {s.value}
   336→          </p>
   337→          <p className="font-mono text-[8.5px] uppercase tracking-wider text-muted-foreground/80">
   338→            {s.label}
   339→          </p>
   340→        </div>
   341→      ))}
   342→    </div>
   343→  );
   344→}

   346→function fmtK(n: number): string {
   347→  return n >= 1000 ? `${(n / 1000).toFixed(n >= 10000 ? 0 : 1)}k` : String(n);
   348→}

   350→/** Calendar-day label for git-log style separators. */
   351→function dayLabel(ts: number): string {
   352→  const d = new Date(ts);
   353→  const now = new Date();
   354→  const startOfDay = (x: Date) => new Date(x.getFullYear(), x.getMonth(), x.getDate()).getTime();
   355→  const diffDays = Math.round((startOfDay(now) - startOfDay(d)) / 86_400_000);
   356→  if (diffDays === 0) return "today";
   357→  if (diffDays === 1) return "yesterday";
   358→  const sameYear = d.getFullYear() === now.getFullYear();
   359→  return d.toLocaleDateString("en-US", {
   360→    month: "short",
   361→    day: "numeric",
   362→    ...(sameYear ? {} : { year: "numeric" }),
   363→  });
   364→}

   366→function LedgerView({
   367→  entries,
   368→  session,
   369→  onPreviewFile,
   370→  running,
   371→}: {
   372→  entries: TimelineEntry[];
   373→  session: Session;
   374→  onPreviewFile: (path: string) => void;
   375→  running: boolean;
   376→}) {
   377→  // newest commit first, like `git log`, with day separators between groups
   378→  const ordered = [...entries].reverse();
   379→
   380→  /** entries the conversation can truncate after — computed once per render */
   381→  const truncatable = React.useMemo(() => {
   382→    const set = new Set<string>();
   383→    for (const e of entries) {
   384→      if (planConversationTruncate(session, e.id)) set.add(e.id);
   385→    }
   386→    return set;
   387→  }, [entries, session]);
   388→
   389→  return (
   390→    <>
   391→      <ol className="relative space-y-1" aria-label="Session activity ledger">
   392→        {/* rail */}
   393→        <span
   394→          aria-hidden
   395→          className="absolute bottom-3 left-[13px] top-3 w-px bg-gradient-to-b from-border via-border to-transparent"
   396→        />
   397→        {ordered.map((e, i) => {
   398→          const prev = i > 0 ? ordered[i - 1] : undefined;
   399→          const newDay = !prev || dayLabel(prev.ts) !== dayLabel(e.ts);
   400→          // stagger caps at 12 rows so long ledgers don't feel sluggish
   401→          const delay = Math.min(i, 12) * 24;
   402→          return (
   403→            <React.Fragment key={e.id}>
   404→              {newDay && (
   405→                <li className="relative z-10 flex items-center gap-2 py-1.5" aria-hidden>
   406→                  <span className="flex size-[27px] shrink-0 items-center justify-center">
   407→                    <GitCommitHorizontal className="size-3.5 text-muted-foreground/40" />
   408→                  </span>
   409→                  <span className="rounded border border-dashed border-border/70 bg-muted/30 px-1.5 py-0.5 font-mono text-[9px] uppercase tracking-wider text-muted-foreground">
   410→                    {dayLabel(e.ts)}
   411→                  </span>
   412→                  <span className="h-px min-w-4 flex-1 bg-border/40" />
   413→                </li>
   414→              )}
   415→              <LedgerRow
   416→                entry={e}
   417→                session={session}
   418→                onPreviewFile={onPreviewFile}
   419→                running={running}
   420→                staggerDelay={delay}
   421→                isHead={i === 0}
   422→                truncatable={truncatable.has(e.id)}
   423→              />
   424→            </React.Fragment>
   425→          );
   426→        })}
   427→      </ol>
   428→      <ShortlogFooter entries={ordered} />
   429→    </>
   430→  );
   431→}

   433→/** `git shortlog` vibes — quiet one-line summary under the ledger rail. */

   435→  const prompts = entries.filter((e) => e.kind === "user").length;
   436→  const replies = entries.filter((e) => e.kind === "assistant").length;
   437→  const calls = entries.filter((e) => e.kind === "tool").length;
   438→  const newest = entries[0];
   439→  if (entries.length === 0) return null;
   440→  return (
   441→    <div
   442→      className="mt-3 flex items-center justify-between gap-2 border-t border-dashed pt-2 font-mono text-[9.5px] text-muted-foreground/60"
   443→      aria-hidden
   444→    >
   445→      <span>
   446→        {prompts} prompt{prompts === 1 ? "" : "s"} · {replies} repl{replies === 1 ? "y" : "ies"} ·{"\n" && " "}
   447→        {calls} tool call{calls === 1 ? "" : "s"}
   448→      </span>
   449→      <span className="flex items-center gap-1">
   450→        <span className="text-sky-400/80">you</span>·<span className="text-[#7c8dfd]/90">dsh</span>
   451→        {newest && (
   452→          <span className="ml-1 opacity-70">@ {hhmm(newest.ts)}</span>
   453→        )}
   454→      </span>
   455→    </div>
   456→  );

   459→function LedgerRow({

   489→        (e.kind === "plan" || e.kind === "todo") && "border-teal-500/50 bg-teal-500/10",

   499→      role={clickable ? "button" : undefined}
   500→      tabIndex={clickable ? 0 : undefined}
   501→      aria-label={clickable ? `Inspect ${e.files![0]} (${e.toolName})` : undefined}
   502→      onKeyDown={
   503→        clickable
   504→          ? (ev) => {
   505→              if (ev.key === "Enter" || ev.key === " ") {
   506→                ev.preventDefault();
   507→                onPreviewFile(e.files![0]);
   508→              }
   509→            }
   510→          : undefined
   511→      }
   512→      onClick={clickable ? () => onPreviewFile(e.files![0]) : undefined}
   513→      className={cn(
   514→        "ml-2 min-w-0 flex-1 rounded-md border border-transparent px-2.5 py-1.5 transition-colors",

   528→            );
   529→          }}
   530→          title="Copy sha"
   531→          className="shrink-0 rounded-sm font-mono text-[10px] tracking-tight text-violet-400/80 underline-offset-2 transition-colors hover:text-violet-300 hover:underline focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring"
   532→        >
   533→          {e.sha}
   534→        </button>

   536→          <span
   537→            aria-label="Current tip of this conversation"
   538→            title="latest commit — HEAD of this conversation"
   539→            className="hidden shrink-0 rounded-sm bg-amber-500/15 px-1 py-px font-mono text-[8.5px] font-bold uppercase tracking-wider text-amber-400 sm:inline-block"
   540→          >
   541→            head
   542→          </span>
   543→        )}

   545→          {meta && (

   638→            onClick={(ev) => {
   639→              ev.stopPropagation();
   640→              setOpen(true);
   641→            }}
   642→            className="rounded-sm p-0.5 text-muted-foreground/0 transition-all hover:!text-amber-400 focus-visible:text-amber-400 focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring group-hover:text-muted-foreground/70"

   644→            <RotateCcw className="size-3" aria-hidden />

   646→        </TooltipTrigger>

   651→        <AlertDialogContent>
   652→          <AlertDialogHeader>
   653→            <AlertDialogTitle className="font-mono">
   654→              rewind workspace to <span className="text-violet-400">{e.sha}</span>?
   655→            </AlertDialogTitle>

   657→              Replays every successful write/edit from the seed tree up to and including this

   663→            <div className="space-y-2 rounded-md border bg-muted/20 p-3 text-xs">

   667→                </p>
   668→              ) : (
   669→                <>
   670→                  {changes.modified.length > 0 && (

   672→                      tone="mod"

   676→                  )}
   677→                  {changes.added.length > 0 && (
   678→                    <ChangeList
   679→                      tone="add"
   680→                      label={`+${changes.added.length} removed from current`}
   681→                      paths={changes.added}
   682→                    />
   683→                  )}

   685→                    <ChangeList
   686→                      tone="del"
   687→                      label={`−${changes.removed.length} missing after rewind`}

   698→                </p>

   701→          )}
   702→
   703→          <AlertDialogFooter>
   704→            <AlertDialogCancel>Cancel</AlertDialogCancel>

   706→              onClick={confirmRestore}
   707→              className="bg-amber-600 text-white hover:bg-amber-600/90"
   708→            >
   709→              <RotateCcw className="mr-1.5 size-3.5" aria-hidden /> Rewind
   710→            </AlertDialogAction>
   711→          </AlertDialogFooter>
   712→        </AlertDialogContent>
   713→      </AlertDialog>
   714→    </>
   715→  );
