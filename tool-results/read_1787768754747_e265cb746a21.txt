     1→/**
     2→ * DSH Web — scripted demo engine.
     3→ *
     4→ * When no DeepSeek API key is configured (or demo mode is forced), this
     5→ * engine stands in for the model: it routes the user's message to one of a
     6→ * few golden-path flows and drives the REAL tool executors against the real
     7→ * virtual workspace — files are genuinely read/written, bash genuinely runs.
     8→ * Only the decision-making is scripted; every tool result you see is real.
     9→ *
    10→ * Emits the same AgentEvent stream as the live agent loop, so the UI is
    11→ * identical (streaming text, tool cards, diffs, todos).
    12→ */

    46→const sleep = (ms: number, signal: AbortSignal) =>
    47→  new Promise<void>((resolve, reject) => {
    48→    const t = setTimeout(resolve, ms);
    49→    signal.addEventListener(
    50→      'abort',
    51→      () => {
    52→        clearTimeout(t);
    53→        reject(new DOMException('Aborted', 'AbortError'));
    54→      },
    55→      { once: true },
    56→    );
    57→  });
    58→
    59→/** Stream text with a lively cadence; aborts cleanly. */
    60→async function streamText(
    61→  text: string,
    62→  onEvent: (e: AgentEvent) => void,
    63→  signal: AbortSignal,
    64→  chunkSize = 6,
    65→  delayMs = 12,
    66→): Promise<void> {
    67→  for (let i = 0; i < text.length; i += chunkSize) {
    68→    if (signal.aborted) throw new DOMException('Aborted', 'AbortError');
    69→    onEvent({ type: 'text-delta', delta: text.slice(i, i + chunkSize) });
    70→    await sleep(delayMs, signal);
    71→  }
    72→}

    74→function fmtNum(n: number): string {

    78→/**
    79→ * Recent web_search results per session, so a follow-up like
    80→ * “fetch the first result” can resolve against the REAL last search.
    81→ * Client-side demo memory only — deliberately ephemeral.
    82→ */
    83→const lastSearchUrls = new Map<string, string[]>();
    84→
    85→const ORDINALS: Array<[RegExp, number]> = [
    86→  [/\b(?:first|top|1st|leading|best)\b/, 0],
    87→  [/\b(?:second|2nd)\b/, 1],
    88→  [/\b(?:third|3rd)\b/, 2],
    89→  [/\b(?:fourth|4th)\b/, 3],
    90→  [/\b(?:fifth|5th)\b/, 4],
    91→];
    92→
    93→function ordinalIndex(text: string): number {
    94→  for (const [re, idx] of ORDINALS) if (re.test(text)) return idx;
    95→  return -1;
    96→}

    98→function extractUrl(raw: string): string | null {
    99→  const m = raw.match(/https?:\/\/[^\s`'"<>）)]+/i);
   100→  if (!m) return null;
   101→  // trim trailing sentence punctuation that often glues onto URLs
   102→  return m[0].replace(/[.,;:!?…]+$/, '');
   103→}
   104→
   105→/**
   106→ * Pull the actual search topic out of a natural-language request:
   107→ * "search the web for deepseek models" → "deepseek models".
   108→ */
   109→function extractSearchQuery(raw: string): string {
   110→  let q = raw.trim();
   111→  q = q.replace(/^(please|can you|could you|dsh[,:]?|hey)\s+/i, '');
   112→  q = q.replace(/^(web\s*search|search\s+(the\s+web|online|the\s+internet))(\s+for|\s+about|\s+:)\s*/i, '');
   113→  q = q.replace(/^look\s*up(\s+on( the)? web)?(\s+for|\s+about)?\s*/i, '');
   114→  q = q.replace(/^search\s+for\s+/i, '');
   115→  q = q.replace(/^(for|about)\s+/i, '');
   116→  q = q.replace(/[?.!]+$/, '');
   117→  return q.length > 0 ? q : raw.trim();
   118→}

   502→      return;
   503→    }
   504→
   505→    /* ------------------------- route: todos / checklist ------------------ */
   506→    if (/(write|make|create|draft) .*(todo|checklist)|plan.*(refactor|release|work)|checklist/.test(text)) {

   541→        const callId = `demo_ask_${Date.now()}`;

   568→          callId,
   569→          name: 'ask_user_question',
   570→          argsRaw: JSON.stringify(askArgs),
   571→        });
   572→        const started = Date.now();
   573→        let answer: string;
   574→        let ok = true;
   575→        try {
   576→          answer = await opts.askUser(askArgs.questions);
   577→        } catch (e) {
   578→          ok = false;
   579→          answer = `Error: ${(e as Error).message}`;
   580→        }
   581→        emitSafe({
   582→          type: 'tool-call-end',
   583→          callId,
   584→          name: 'ask_user_question',

   592→          signal,

   598→          signal,

   601→      emitSafe({ type: 'done', aborted: false });

   603→    }
   604→
   605→    /* ------------------------- default: capabilities --------------------- */
   606→    const files = opts.listFiles();
   607→    const bytes = Object.values(opts.listFiles()).length;
   608→    void bytes;
   609→    const intro = `**Demo mode** — no DeepSeek API key is configured, so I'm running on a scripted engine. Tool execution is still 100% real: everything happens against the virtual workspace in your browser (${fmtNum(files.length)} files currently tracked).

   612→

   618→- **“search the web for deepseek models”** — real \`web_search\` via the server-side plugin (works even in demo)
   619→- **“fetch the first result”** — real \`web_fetch\` reads the page behind your last search's top hit

   623→Then paste your DeepSeek API key in **Settings → Models** to unlock the full agent loop with \`deepseek-chat\` / \`deepseek-reasoner\`.`;
   624→    await streamText(intro, emitSafe, signal);
   625→    emitSafe({ type: 'done', aborted: false });
   626→  } catch (e) {
   627→    if (signal.aborted) {
   628→      emitSafe({ type: 'done', aborted: true });
   629→      return;
   630→    }
   631→    emitSafe({ type: 'error', message: `Demo engine failure: ${(e as Error).message}` });
   632→    emitSafe({ type: 'done', aborted: false });
   633→  }
   634→}
